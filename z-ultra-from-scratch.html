1. Drawing Fullscreen UI with Touch Input (Python + LVGL / evdev) Screen & Touch 
# app_screen_touch.py - Minimal Fullscreen UI directly on /dev/graphics/fb0
import evdev
import mmap
import time

FB_NODE = "/dev/graphics/fb0"
WIDTH, HEIGHT, BPP = 1080, 1920, 4  # Xperia Z Ultra FHD panel

def render_color_bar():
    # Direct mmap framebuffer write - zero X11/Wayland overhead!
    with open(FB_NODE, "r+b") as f:
        fb = mmap.mmap(f.fileno(), WIDTH * HEIGHT * BPP)
        # Fill screen with vibrant deep cyan (ARGB: 0xFF008080)
        cyan_pixel = b'\x80\x80\x00\xFF' * (WIDTH * HEIGHT)
        fb.seek(0)
        fb.write(cyan_pixel)

def listen_touch():
    touch_dev = evdev.InputDevice('/dev/input/event1')
    print(f"[+] Listening for touch events on {touch_dev.name}...")
    for event in touch_dev.read_loop():
        if event.type == evdev.ecodes.EV_ABS:
            absevent = evdev.categorize(event)
            if event.code == evdev.ecodes.ABS_MT_POSITION_X:
                x = event.value
            elif event.code == evdev.ecodes.ABS_MT_POSITION_Y:
                y = event.value
                print(f"[Touch] X: {x}, Y: {y}")

if __name__ == "__main__":
    render_color_bar()
    listen_touch()
2. Reading 6-Axis Gyroscope & Accelerometer (InvenSense MPU6500) Gyro & Accel 
# app_sensors.py - Read 6-Axis motion vector from Linux IIO sysfs
import time

IIO_PATH = "/sys/bus/iio/devices/iio:device0"

def read_sensor_raw(node_name):
    try:
        with open(f"{IIO_PATH}/{node_name}", "r") as f:
            return float(f.read().strip())
    except FileNotFoundError:
        return 0.0

def get_motion_data():
    # Angular velocity (degrees/sec)
    gyro_x = read_sensor_raw("in_anglvel_x_raw")
    gyro_y = read_sensor_raw("in_anglvel_y_raw")
    gyro_z = read_sensor_raw("in_anglvel_z_raw")

    # Linear acceleration (m/s^2)
    accel_x = read_sensor_raw("in_accel_x_raw")
    accel_y = read_sensor_raw("in_accel_y_raw")
    accel_z = read_sensor_raw("in_accel_z_raw")

    return {
        "gyro": (gyro_x, gyro_y, gyro_z),
        "accel": (accel_x, accel_y, accel_z)
    }

while True:
    data = get_motion_data()
    print(f"Gyro: {data['gyro']} | Accel: {data['accel']}")
    time.sleep(0.1)
